📚 什么是多态？

多态 (Polymorphism) = 多种形态

同一个行为，不同的对象有不同的表现方式！

🎭 生活中的多态例子

🐕
狗
叫声：汪汪汪
🐱
猫
叫声：喵喵喵
🐦
鸟
叫声：叽叽喳喳

💡 核心思想：都是"叫"这个动作，但每种动物的实现方式不同！

😕 传统C语言的困境

❌ 如果用传统的if-else或switch来处理不同类型...

// 糟糕的实现方式 😰 void makeSound(int animalType) { switch(animalType) { case 1: // 狗 printf("汪汪汪\n"); break; case 2: // 猫 printf("喵喵喵\n"); break; case 3: // 鸟 printf("叽叽喳喳\n"); break; // 每增加一种动物，都要修改这个函数！😱 } }

😣 问题1：扩展性差

每次添加新类型都要修改代码

😵 问题2：维护困难

代码会越来越臃肿

🤯 问题3：违反开闭原则

对修改开放，对扩展封闭

🦸 函数指针来救场！

✨ 函数指针：可以指向函数的指针变量

就像普通指针指向数据，函数指针指向代码！

void (*ptr)()
📦 函数代码
printf("Hello!");
// 声明函数指针类型 typedef void (*SoundFunc)(); // 不同的实现函数 void dogSound() { printf("汪汪汪\n"); } void catSound() { printf("喵喵喵\n"); } // 使用函数指针 SoundFunc soundPtr = dogSound; // 指向狗叫函数 soundPtr(); // 调用：输出 "汪汪汪" soundPtr = catSound; // 改为指向猫叫函数 soundPtr(); // 调用：输出 "喵喵喵"

💡 关键优势：运行时动态决定调用哪个函数！

🏗️ 构建对象结构

🎯 将数据和行为封装在一起

// 定义Animal结构体 typedef struct { char name[20]; void (*makeSound)(); // 函数指针：叫声 void (*move)(); // 函数指针：移动 } Animal; // 创建不同的动物 Animal dog = {"小黄", dogSound, dogMove}; Animal cat = {"咪咪", catSound, catMove}; // 统一的调用方式！ dog.makeSound(); // 汪汪汪 cat.makeSound(); // 喵喵喵

🎉 完整的多态实现

#include <stdio.h> #include <string.h> // 基础Animal结构 typedef struct Animal { char name[20]; void (*makeSound)(struct Animal*); void (*move)(struct Animal*); } Animal; // Dog的具体实现 void dogSound(Animal* self) { printf("%s说：汪汪汪！\n", self->name); } void dogMove(Animal* self) { printf("%s在快速奔跑！\n", self->name); } // Cat的具体实现 void catSound(Animal* self) { printf("%s说：喵喵喵～\n", self->name); } void catMove(Animal* self) { printf("%s在优雅地走路\n", self->name); } // 创建动物的工厂函数 Animal createDog(const char* name) { Animal dog; strcpy(dog.name, name); dog.makeSound = dogSound; dog.move = dogMove; return dog; } Animal createCat(const char* name) { Animal cat; strcpy(cat.name, name); cat.makeSound = catSound; cat.move = catMove; return cat; } // 多态的威力：统一处理不同动物 void performActions(Animal* animal) { animal->makeSound(animal); // 多态调用 animal->move(animal); // 多态调用 } int main() { Animal dog = createDog("旺财"); Animal cat = createCat("小花"); // 使用同样的函数处理不同的动物 performActions(&dog); performActions(&cat); return 0; }

🎯 运行结果：

旺财说：汪汪汪！ 旺财在快速奔跑！ 小花说：喵喵喵～ 小花在优雅地走路

🚀 虚函数表 - 更优雅的实现

💡 将所有函数指针集中管理，模拟C++的虚函数表

📋 Dog虚函数表

makeSound → dogSound()
move → dogMove()
eat → dogEat()
// 虚函数表结构 typedef struct { void (*makeSound)(void* self); void (*move)(void* self); void (*eat)(void* self); } VTable; // 带虚函数表的Animal typedef struct { const VTable* vtable; // 指向虚函数表 char name[20]; } Animal; // Dog的虚函数表 static const VTable dogVTable = { .makeSound = dogSound, .move = dogMove, .eat = dogEat }; // Cat的虚函数表 static const VTable catVTable = { .makeSound = catSound, .move = catMove, .eat = catEat }; // 使用宏简化调用 #define CALL(obj, method, ...) ((obj)->vtable->method(obj, ##__VA_ARGS__)) // 优雅的多态调用 void animalLife(Animal* animal) { CALL(animal, makeSound); CALL(animal, move); CALL(animal, eat); }
