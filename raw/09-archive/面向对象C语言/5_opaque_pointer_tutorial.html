💻 完整代码

📄 door.h
#ifndef DOOR_H
#define DOOR_H
#include <stdbool.h>

typedef struct Door Door;

Door* door_create(int password);
bool door_open(Door* d, int pwd);
void door_close(Door* d);
void door_destroy(Door* d);

#endif
📄 door.c
#include "door.h"
#include <stdlib.h>

struct Door {
    int password;
    int open_count;
    bool is_locked;
};

Door* door_create(int password) {
    Door* d = malloc(sizeof(Door));
    d->password = password;
    d->open_count = 0;
    d->is_locked = true;
    return d;
}

bool door_open(Door* d, int pwd) {
    if (d->password == pwd) {
        d->is_locked = false;
        d->open_count++;
        return true;
    }
    return false;
}

void door_close(Door* d) {
    d->is_locked = true;
}

void door_destroy(Door* d) {
    free(d);
}
📄 main.c
#include "door.h"
#include <stdio.h>

int main() {
    Door* d = door_create(1234);
    
    if (door_open(d, 1234)) {
        printf("门开了!\n");
    }
    
    door_close(d);
    door_destroy(d);
    
    return 0;
}
