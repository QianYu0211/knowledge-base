typedef struct { // 硬件资源 USART_TypeDef* hw; // 寄存器基地址 // 配置参数 uint32_t baudrate; // 波特率 // 缓冲区 uint8_t tx_buffer[256]; // 发送缓冲 uint8_t rx_buffer[256]; // 接收缓冲 } UART_Device;
💡 关键理解: struct就像一个容器,把UART需要的所有数据打包在一起。每创建一个对象,就有一份独立的数据副本,互不干扰!

🤔 思考: 数据有了,但怎么让对象"动起来"?

typedef struct UART_Device UART_Device; struct UART_Device { // 数据成员 USART_TypeDef* hw; uint32_t baudrate; // 方法成员(函数指针) void (*init)(UART_Device* self); void (*send)(UART_Device* self, const char* data, int len); };
🔑 语法重点: void (*init)(...) 括号很重要! 每个函数指针的第一个参数都是 UART_Device* self - 这就是接下来要讲的 this指针!

👆 第三步:this指针 - 方法的灵魂

🎯 核心问题: 当调用 uart1->send() 时,
send函数怎么知道要操作的是uart1还是uart2?

答案:this指针!
uart1 对象
地址: 0x20000100
hw = USART1
baudrate = 115200
send 方法
收到参数:
self = 0x20000100

操作:
self->hw->DR = data[0]
// 当你调用: uart1->send(uart1, "Hello", 5); ^^^^^^ 这就是this指针! // 函数内部通过self访问对象成员 void uart_send_impl(UART_Device* self, const char* data, int len) { self->hw->DR = data[0]; // 操作这个对象的硬件 memcpy(self->tx_buffer, data, len); }
🏦 生活化比喻: 就像你去ATM取款,必须先插卡(传入this指针),告诉机器"我是张三"。ATM才知道从张三的账户取钱。
this指针 = 对象的身份证!

UART_Device* UART_Device_Create( USART_TypeDef* hw, uint32_t baud) { // 1️⃣ 分配内存 UART_Device* dev = malloc(sizeof(UART_Device)); // 2️⃣ 初始化数据成员 dev->hw = hw; dev->baudrate = baud; // 3️⃣ 绑定方法(关键!) dev->init = uart_init_impl; dev->send = uart_send_impl; return dev; }
🔧 构造函数三部曲:
1️⃣ 分配内存 - 给对象找个"家"
2️⃣ 初始化数据 - 填写"身份信息"
3️⃣ 绑定方法 - 安装"遥控按钮"(最关键!)

⚙️ 方法实现:init 和 send

void uart_init_impl(UART_Device* self) { // 通过self访问对象的硬件 self->hw->CR1 |= USART_CR1_UE; // 根据对象的波特率配置 uint32_t div = 42000000/(16*self->baudrate); self->hw->BRR = div; // 配置收发使能 self->hw->CR1 |= USART_CR1_TE | USART_CR1_RE; }
void uart_send_impl(UART_Device* self, const char* data, int len) { for(int i=0; i// 等待发送寄存器空 while(!(self->hw->SR & USART_SR_TXE)); // 写入数据 self->hw->DR = data[i]; } }
💡 关键理解:
• 所有操作都通过 self-> 访问对象成员
• 同一个函数,通过不同的self参数可以操作不同对象
• uart1->send() 和 uart2->send() 调用同一函数,但操作不同硬件!

这就是OOP的精髓:代码复用 + 数据隔离
