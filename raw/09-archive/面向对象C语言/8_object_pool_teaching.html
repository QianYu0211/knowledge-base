Object* pool_alloc(ObjectPool* pool) {
    for (int i = 0; i < pool->pool_size; i++) {
        if (pool->pool[i].is_free == 1) {
            pool->pool[i].is_free = 0;

        }
    }
    return NULL;
}

💻 完整代码实现

#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int is_free;
    char data[256];
} Object;

typedef struct {
    Object* pool;
    int pool_size;
    int used_count;
} ObjectPool;

ObjectPool* pool_init(int size) {
    ObjectPool* p = (ObjectPool*)malloc(sizeof(ObjectPool));
    p->pool = (Object*)malloc(size * sizeof(Object));
    p->pool_size = size;
    p->used_count = 0;
    for (int i = 0; i < size; i++) {
        p->pool[i].is_free = 1;
    }
    return p;
}

Object* pool_alloc(ObjectPool* p) {
    for (int i = 0; i < p->pool_size; i++) {
        if (p->pool[i].is_free) {
            p->pool[i].is_free = 0;
            p->used_count++;
            return &p->pool[i];
        }
    }
    return NULL;
}

void pool_free(ObjectPool* p, Object* obj) {
    if (obj >= p->pool && obj < p->pool + p->pool_size) {
        obj->is_free = 1;
        p->used_count--;
    }
}

void pool_destroy(ObjectPool* p) {
    free(p->pool);
    free(p);
}

int main() {
    ObjectPool* pool = pool_init(5);
    Object* obj1 = pool_alloc(pool);
    Object* obj2 = pool_alloc(pool);
    printf("已用: %d/5\n", pool->used_count);
    pool_free(pool, obj1);
    printf("已用: %d/5\n", pool->used_count);
    pool_destroy(pool);
    return 0;
}
