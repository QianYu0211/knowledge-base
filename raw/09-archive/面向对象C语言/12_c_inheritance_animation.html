📦 方法一：结构体嵌套

💡 核心思想：把"父类"作为"子类"的第一个成员！ 
// 定义"父类" - 基础车辆
typedef struct {
    char brand[50];    // 品牌
    char color[20];    // 颜色
    int speed;          // 速度
} Vehicle;

// 定义"子类" - 跑车
typedef struct {
    Vehicle base;       // 关键！把父类作为第一个成员
    int maxSpeed;       // 跑车特有：最高时速
    int acceleration;   // 跑车特有：加速性能
} SportsCar;

🎮 方法二：函数指针实现方法继承

💡 核心思想：用函数指针模拟"虚函数"，实现方法的继承和重写！ 
// 定义带有"方法"的父类
typedef struct Vehicle {
    char brand[50];
    char color[20];
    int speed;
    
    // 函数指针 - 相当于"方法"
    void (*drive)(struct Vehicle* self);
    void (*display)(struct Vehicle* self);
} Vehicle;

// 定义子类
typedef struct {
    Vehicle base;          // 继承父类
    int turboMode;         // 子类特有属性
    
    // 子类可以重写父类的方法！
} SportsCar;

🔧 函数指针工作原理

父类方法： vehicle->drive(vehicle)
⬇️
子类重写： sportsCar->base.drive = sportsCar_drive
⬇️
多态调用： 同样的接口，不同的实现！
场景 4/5

🦁 完整示例：动物继承体系

// 基类：动物
typedef struct Animal {
    char name[30];
    int age;
    void (*makeSound)(struct Animal* self);
} Animal;

// 派生类：狗
typedef struct {
    Animal base;        // 继承动物
    char breed[30];     // 狗的品种
} Dog;

// 派生类：猫
typedef struct {
    Animal base;        // 继承动物
    int lives;          // 猫有九条命
} Cat;

// 实现狗的叫声
void dog_sound(Animal* self) {
    printf("%s: 汪汪汪！\n", self->name);
}

// 实现猫的叫声
void cat_sound(Animal* self) {
    printf("%s: 喵喵喵！\n", self->name);
}

// 使用示例
int main() {
    Dog myDog = {{"旺财", 3, dog_sound}, "柴犬"};
    Cat myCat = {{"咪咪", 2, cat_sound}, 9};
    
    // 多态调用
    Animal* animals[] = {(Animal*)&myDog, (Animal*)&myCat};
    
    for(int i = 0; i < 2; i++) {
        animals[i]->makeSound(animals[i]);  // 多态！
    }
}

🐱 Cat 对象

🎊 总结：C语言继承的核心技巧

✅ 优点

  • 代码复用性高
  • 结构清晰，易于维护
  • 可以实现多态
  • 内存布局可控

  • 类型转换需谨慎
  • 函数指针初始化别忘记
  • 内存对齐要注意
  • 没有编译器的类型检查
