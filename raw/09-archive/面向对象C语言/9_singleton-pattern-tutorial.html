📦 完整实现

// 结构体定义
typedef struct {
    int port;
    int pageCount;
} Printer;

// 单例获取函数
Printer* getPrinterInstance() {
    static Printer instance;
    static int initialized = 0;
    
    if (!initialized) {
        instance.port = 0x378;
        instance.pageCount = 0;
        initialized = 1;
    }
    
    return &instance;
}

// 使用示例
int main() {
    Printer* p1 = getPrinterInstance();
    Printer* p2 = getPrinterInstance();
    
    // p1 == p2 ✅ 同一个对象
    return 0;
}
